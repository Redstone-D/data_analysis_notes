\section{Least Squares Error}

\begin{eg}[Comparing Two Straight-Line Fits]{straight-line-squared-error}
    Consider the three observations
    \[
      \begin{array}{c|ccc}
        i & 0 & 1 & 2 \\
        \hline
        x(i) & 0 & 1 & 2 \\
        y(i) & 1 & 2 & 2
      \end{array}
    \]
    Let $c_A,c_B:\RealVectors{2}$ have coordinates
    \[
      (c_A(0),c_A(1))=(1,1),\qquad
      (c_B(0),c_B(1))=\left(1,\frac12\right)
    \]
    \begin{enumerate}
      \item State $n$ and $m$, and write the functions generated by $c_A$
        and $c_B$
      \item Define $S\defeq\PolyError(n)(m)(x)(y)$, state its signature,
        and expand $S(c)$ into a sum of squared differences
      \item Calculate $S(c_A)$ and $S(c_B)$. Which candidate line has the
        smaller sum of squared errors?
      \item Use the
        \xref[least-squares derivative criterion]{lec02::thm:straight-line-least-squares-equations}
        to find a coefficient vector $c^*:\RealVectors{2}$ minimizing $S$
        over all straight-line models, and write its fitted function
    \end{enumerate}
    \begin{enumerate}
      \item $n = 3$ and $m = 1$. The functions generated by $c_A$ and $c_B$ are:
        \[
          f_A(t) = c_A(0) + c_A(1) \cdot t = 1 + 1 \cdot t = t + 1
        \]
        \[
          f_B(t) = c_B(0) + c_B(1) \cdot t = 1 + \frac{1}{2} \cdot t = \frac{t}{2} + 1
        \]
      \item $S\defeq\PolyError(n)(m)(x)(y)$
      \[
        \begin{aligned}
          \PolyError &: \Pi n:\NNOne.\,\Pi m:\NN.\,(\Fin(n)\to\RR) \\
          &\qquad\to(\Fin(n)\to\RR)\to(\RealVectors{m+1}\to\RR)
        \end{aligned}
      \]
      \[
        n:\NNOne=3,\qquad m:\NN=1
      \]
      $$\vdash \PolyError(3)(1): (\Fin(3)\to\RR) \to(\Fin(3)\to\RR)\to(\RealVectors{2}\to\RR)$$

      Notice that $x, y: \Fin(3) \to \RR$, so we know that
      $$S = \PolyError(3)(1)(x)(y): \RealVectors{2}\to\RR$$
      \item For any coefficient vector $c:\RealVectors{2}$,
      \begin{align*}
        S(c) &= \SquaredError(3)(x)(y)(\PolynomialModelGen(1)(c)) \\
        &= \sum_0^2\left(i\map
          \dist{y(i)}{\PolynomialModelGen(1)(c)(x(i))}^2\right) \\
        &= \sum_0^2\left(i\map
          \abs{y(i)-\bigl(c(0)+c(1)\cdot x(i)\bigr)}^2\right)
      \end{align*}

      For $c_A=(1,1)$,
      \begin{align*}
        S(c_A) &= \sum_0^2\left(i\map\abs{y(i)-1-x(i)}^2\right) \\
        &= \abs{1-1-0}^2+\abs{2-1-1}^2+\abs{2-1-2}^2 \\
        &= 0^2+0^2+(-1)^2 \\
        &= 1
      \end{align*}

      For $c_B=\left(1,\frac12\right)$,
      \begin{align*}
        S(c_B) &= \sum_0^2\left(i\map
          \abs{y(i)-1-\frac12\cdot x(i)}^2\right) \\
        &= \abs{1-1-\frac12\cdot0}^2
          +\abs{2-1-\frac12\cdot1}^2
          +\abs{2-1-\frac12\cdot2}^2 \\
        &= 0^2+\left(\frac12\right)^2+0^2 \\
        &= \frac14
      \end{align*}

      Since $S(c_B)=\frac14<1=S(c_A)$, candidate $B$ has the smaller
      sum of squared errors
      \item Construct
        \[
          S\defeq c\map
            (c(0)-1)^2+(c(0)+c(1)-2)^2+(c(0)+2\cdot c(1)-2)^2
        \]
        We require $\PartialD_0(S)(c)=\PartialD_1(S)(c)=0$.
        Differentiating gives
        \begin{align*}
          \PartialD_0(S)(c)
            &=2\cdot(c(0)-1)+2\cdot(c(0)+c(1)-2)\\
            &\qquad+2\cdot(c(0)+2\cdot c(1)-2)\\
            &=2\cdot(3\cdot c(0)+3\cdot c(1)-5),\\
          \PartialD_1(S)(c)
            &=2\cdot(c(0)+c(1)-2)+4\cdot(c(0)+2\cdot c(1)-2)\\
            &=2\cdot(3\cdot c(0)+5\cdot c(1)-6)
        \end{align*}
        Setting both derivatives to zero and dividing by $2$ gives
        \[
          \begin{cases}
            3\cdot c(0)+3\cdot c(1)=5\\
            3\cdot c(0)+5\cdot c(1)=6
          \end{cases}
        \]
        Subtracting the first equation from the second gives
        $2\cdot c(1)=1$, so $c(1)=\frac12$.
        Substituting into the first equation gives $c(0)=\frac76$.
        Thus the minimizing vector and its fitted function are
        \[
          c^*=\left(\frac76,\frac12\right),\qquad
          \PolynomialModelGen(1)(c^*)=t\map\frac76+\frac12\cdot t
        \]
        Its squared error is $S(c^*)=\frac16<\frac14=S(c_B)$
    \end{enumerate}
\end{eg}
