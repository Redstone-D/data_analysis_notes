\section{Data Analysis in Science} 

\begin{dfn}[Analytical Method and Analytical Solution]{analytical-method}
  When we use \emph{analytical methods} to solve a problem, we are manipulating the mathematical expressions until unknowns are being isolated into a standard mathematical form. 

  An \emph{analytical solution} is a solution for a problem such that the unknowns are expressed in a closed form. 
\end{dfn}

\begin{eg}[Analytical Solution of a Quadratic Equation]{quadratic-equation}
  Consider the quadratic equation
  \begin{equation}
    ax^2 + bx + c = 0,
  \end{equation}
  where $a$, $b$, and $c$ are real constants and $a\ne0$.
  When $b^2-4ac\geq0$, the real solutions are given by the quadratic formula:
  \begin{equation}
    x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}.
  \end{equation} 
\end{eg}

\begin{dfn}[Numerical Method and Numerical Solution]{numerical-method}
  \emph{Numerical methods} are methods to obtain numerical solutions. 

  A \emph{numerical solution} is a solution for a problem such that the unknowns are expressed in an numerical form.
\end{dfn} 

\begin{eg}[Numerical Solution of a Trigonometric Equation]{trig-equation}
  Consider the trigonometric equation
  \begin{equation}
    x = \beta \cos x
  \end{equation} 
  where $\beta$ is a constant. 

  It is very difficult to find an analytical solution for this equation using algebraic manipulations. However, we can use numerical methods to find an approximate solution for specific $\beta$ values. 

  The numerical solutions can be 
  \[
  \beta = 1, \quad x \approx 0.73909, \quad \beta = 0.5, \quad x \approx 0.45018
  \]
\end{eg} 

\begin{rem}[Advantages and Disadvantages of Numerical Methods]{numerical-methods}
  \begin{enumerate}[nosep] 
    \item A numerical run uses specified input values; unknown parameters may themselves be quantities to estimate
    \item Changing the input values may require another run
    \item Numerical solutions are generally approximate; their accuracy depends on the method and stopping rule
    \item Analytical formulas can explain parameter dependence, while numerical methods can provide useful values even when a closed form exists
  \end{enumerate}

\end{rem}

\section{Fixed Point Iteration Method} 

\begin{dfn}[Fixed Point Iteration Method]
  The idea for \emph{fixed point iteration method} is to first find a function $g(x)$ such that root $r$ is a \xref[fixed point]{fundamental-computing::lec07::def:fixed-point} of $g(x)$, i.e., $g(r) = r$
\end{dfn}

\begin{conv}[Procedure of Fixed Point Iteration Method]{fixed-point-iteration}
  The classical procedure of fixed point iteration method is as follows: 
  \begin{enumerate}[nosep]
    \item Find a function $g(x)$ such that root $r$ is a fixed point of $g(x)$, i.e., $g(r) = r$.
    \item Choose an initial guess $x_0$.
    \item Choose a maximum number of iterations $N$ 
    \item Iterate $x_{n+1} = g(x_n)$ for $N$ times 
  \end{enumerate} 
\end{conv}

\begin{rem}[Error Analysis of Fixed Point Iteration Method]{fixed-point-iteration-error}
  Successive iterates need not lie on opposite sides of a fixed point,
  and their distance alone is not a bound on the root error

  Suppose $g$ is differentiable on an open interval containing a closed
  interval $I=[a,b]$, where $a<b$, and
  \[
    g(I)\subseteq I,\qquad x_0:I,\qquad
    \exists q:\RR,\quad 0\leq q<1,\quad
    \forall t:I,\quad |g'(t)|\leq q
  \]
  Then $x_{n+1}=g(x_n)$ converges to the unique fixed point $r:I$, and
  \[
    |x_{n+1}-r|\leq q\cdot|x_n-r|,\qquad
    |x_{n+1}-r|\leq\frac{q}{1-q}\cdot|x_{n+1}-x_n|
  \]
  Thus a certified root-error tolerance $\varepsilon>0$ is met when
  $\frac{q}{1-q}\cdot|x_{n+1}-x_n|\leq\varepsilon$
\end{rem}

\begin{proof}[Convergence Intuition]
  \leavevmode\par
  \missingprereq{Mean value theorem and completeness of the real numbers;
  real-analysis}
  \begin{itemize}[nosep]
    \item The derivative bound shrinks distances by at most a factor $q$
    \item Successive changes are bounded by a geometric sequence, so the
      iterates converge within the closed interval; continuity makes
      their limit a fixed point
    \item Summing the remaining changes gives the displayed root-error
      bound. Two distinct fixed points would contradict the distance bound
  \end{itemize}
\end{proof}

\begin{conv}[Optimized Fixed Point Iteration]{optimized-fpi}
  Set a step tolerance $\mathrm{tol}>0$ and stop when
  $|x_{n+1}-x_n|<\mathrm{tol}$. This test alone does not certify a root
  error below $\mathrm{tol}$; use the assumptions and bound in the
  \xref[fixed-point error analysis]{rem:fixed-point-iteration-error}
    \begin{enumerate}[nosep]
    \item Find a function $g(x)$ such that root $r$ is a fixed point of $g(x)$, i.e., $g(r) = r$.
    \item Choose an initial guess $x_0$.
    \item Choose a maximum number of iterations $N$ 
    \item Set an error tolerance, $\mathrm{tol}$ 
    \item Iterate $x_{n+1} = g(x_n)$ 
    \item Check if $|x_{n+1} - x_n| < \mathrm{tol}$, if yes, stop the iteration and return the value. 
    \item If $N$ updates are exhausted without meeting the stopping test, stop with an error
  \end{enumerate}

\end{conv}

\section{Bisector Method}

\begin{dfn}[Bisector Method]{bisector-method}
  The idea of \emph{bisector method} is when we know that a root $r$ is in the interval $[a, b]$, then we can find the midpoint of the range and evaluate which part of the interval contains the root. We can then repeat this process until we find a sufficiently accurate approximation of the root. 
\end{dfn} 

\begin{conv}[Procedure of Bisector Method]{bisector-method-procedure}
  Let $f$ be continuous on $[a_0,b_0]$, where $a_0<b_0$
  \begin{enumerate}[nosep]
    \item If either endpoint is an exact root, return it. Otherwise require
      $f(a_0)\cdot f(b_0)<0$
    \item Choose $N:\NN$ halvings and initialize $a=a_0$, $b=b_0$
    \item For each halving, compute $c=(a+b)/2$. If $f(c)=0$, return $c$
    \item Otherwise, if $f(a)\cdot f(c)<0$, set $b=c$; if not, set $a=c$
    \item Repeat until $N$ halvings are complete, then return the midpoint
      $(a+b)/2$ of the final interval
  \end{enumerate}
  Alternatively, stop once half the current interval width is at most the
  requested absolute error, and return its midpoint
\end{conv}

\begin{rem}[Error Analysis of Bisector Method]{bisector-method-error}
  Unless an exact root is found earlier, after $k$ halvings the retained
  interval $[a_k,b_k]$ contains a root and has width
  \[
    b_k-a_k=\frac{b_0-a_0}{2^k}
  \]
  Its midpoint $c_k=(a_k+b_k)/2$ therefore satisfies, for a root
  $r:[a_k,b_k]$,
  \[
    |c_k-r|\leq\frac{b_k-a_k}{2}
      =\frac{b_0-a_0}{2^{k+1}}
  \]
  Returning an endpoint instead gives the weaker bound $b_k-a_k$.
  Choose $k$ so that the appropriate bound is at most the requested
  tolerance; the width is an error bound, not the exact error
\end{rem} 

\section{Newton-Raphson Method} 

\begin{definition}[Newton-Raphson Method]{newton-raphson-method}
  The idea of \emph{Newton-Raphson method} is to use the slope of the function at a point to find the root of the function
\end{definition}

\begin{rem}[Intuition of Newton-Raphson Method]{newton-raphson-intuition}
  At the current guess $x_n$, approximate $f$ by its tangent line:
  \[y = f'(x_n)\,(x - x_n) + f(x_n).\]
  The next guess $x_{n+1}$ is where this tangent line crosses zero. Setting $y = 0$ and solving for $x$:
  \[0 = f'(x_n)\,(x - x_n) + f(x_n) \quad\Longrightarrow\quad x = x_n - \frac{f(x_n)}{f'(x_n)}.\]
  This gives the update rule used in step 4 below.
\end{rem}

\begin{conv}[Procedure of Newton-Raphson Method]{newton-raphson-method-procedure}
  \begin{enumerate}[nosep]
    \item Choose an initial guess $x_0$, a tolerance $\mathrm{tol}>0$, and an iteration limit $N$
    \item If $f(x_n)=0$, return $x_n$
    \item Evaluate $f'(x_n)$; if it is zero away from a root, stop with an error
    \item Update:
      \[x_{n+1} = x_n - \frac{f(x_n)}{f'(x_n)}\]
    \item If $f(x_{n+1})=0$ or $|x_{n+1}-x_n|<\mathrm{tol}$, return $x_{n+1}$
    \item Repeat from step 2; if $N$ updates are exhausted, stop with an error
  \end{enumerate}
  Reject non-finite function values, derivatives and iterates
\end{conv}

\begin{rem}[Newton-Raphson Error Analysis]{newton-raphson-method-error}
  Let $f(r)=0$, $f'(r)\ne0$, and $e_n=x_n-r$.
  Assume \missingprereq{twice continuous differentiability near $r$, real-analysis}.
  A \xref[Taylor expansion with remainder]{lec03::thm:finite-taylor-expansion}
  through the linear term gives
  \[
    0=f(x_n)-f'(x_n)\cdot e_n
      +\frac{f''(\xi_n)}{2}\cdot e_n^2
  \]
  for some $\xi_n$ between $x_n$ and $r$. Hence
  \[
    e_{n+1}=e_n-\frac{f(x_n)}{f'(x_n)}
      =\frac{f''(\xi_n)}{2f'(x_n)}\cdot e_n^2
      \approx\frac{f''(r)}{2f'(r)}\cdot e_n^2
  \]
  This is local quadratic convergence when the initial guess is sufficiently
  close to $r$. The stopping test $|x_{n+1}-x_n|<\mathrm{tol}$ alone does not
  certify $|x_{n+1}-r|<\mathrm{tol}$; also inspect the residual $|f(x_{n+1})|$
\end{rem}
